\section{Overview}

\begin{frame}{Overview: qué hicimos y qué aprendimos}
\begin{columns}[T]
\column{0.45\textwidth}
\begin{block}{Modelo ganador}
\small
\textbf{XGBoost por hogar} (163 variables) alimentado por modelos por persona
(ingreso con pérdida \textbf{Tweedie}, probabilidad de pobreza) y combinado
70/30 con una regresión del ingreso. Umbral 0,39.
\end{block}
\vspace{2mm}
\centering\small
\begin{tabular}{@{}lc@{}}
\toprule
$F_1$ validación cruzada (5 folds) & \textbf{0,7775} \\
$F_1$ con umbral honesto & 0,7775 \\
\textbf{Kaggle (public leaderboard)} & \textbf{0,7775} \\
\bottomrule
\end{tabular}

\column{0.52\textwidth}
\textbf{Takeaways}
\begin{enumerate}\small
  \item Con IA, \alert{combinar variables ya no diferencia}: todos los grupos
    llegan a $F_1$ parecidos con los mismos datos.
  \item Ganamos con tres pilares: \pval{} \pher{} \pecon{}
  \item La pobreza es \textbf{del hogar} y la explica el \textbf{trabajo}:
    formalidad, ocupación e ingresos no laborales.
  \item Educación, salud y subsidios casi no predicen el ingreso: pobreza
    monetaria $\neq$ multidimensional.
\end{enumerate}
\end{columns}
\end{frame}

\section{Datos}
% =============================================================================

\begin{frame}{Datos: por qué estas variables}
\small
GEIH 2018 (DANE): \textbf{543.103 personas, 164.958 hogares}, 20,0\% de hogares
pobres. Pobre $\iff$ ingreso per cápita de la unidad de gasto \emph{con
arriendo imputado} $<$ línea de pobreza del dominio.
\vspace{1mm}

\footnotesize
\begin{tabular}{@{}p{2.9cm}p{7.2cm}p{3.6cm}@{}}
\toprule
\textbf{Bloque} & \textbf{Mecanismo económico} & \textbf{Sustento} \\
\midrule
Mercado laboral & El trabajo es la principal fuente de ingreso; la
  informalidad separa ingresos estables de precarios & MESEP (2012);
  Fernández y Villar (2017) \\
Ingresos no laborales & Pensiones, arriendos, remesas, ayudas: ingreso sin
  estar ocupado & DANE (2019) \\
Vivienda y arriendo & El arriendo imputado \emph{entra} en la definición de
  ingreso del DANE & MESEP (2012) \\
Demografía & El mismo ingreso, dividido entre más miembros y menores &
  Ravallion (2016) \\
Capital humano y salud & Retornos a la educación; régimen de salud como proxy
  de formalidad & Angulo et al.\ (2016) \\
\bottomrule
\end{tabular}
\vspace{2mm}

\pecon{} \small Ningún monto de ingreso está en test: el modelo
\emph{reconstruye} el ingreso con las preguntas que sí están.
\end{frame}

\begin{frame}{Entrenamiento: una validación concienzuda \hfill \pval}
\begin{columns}[T]
\column{0.52\textwidth}
\small
\begin{itemize}
  \item \textbf{Folds por hogar} (5): todos los miembros en el mismo fold.
  \item \textbf{CV anidada}: el ingreso predicho y la probabilidad por persona
    usan la etiqueta, así que se estiman \alert{sin ver} el fold evaluado.
  \item \textbf{Parada temprana}: 100 rondas sin mejorar la log-loss.
  \item \textbf{Umbral} escogido por $F_1$ (grilla 0,10--0,85).
\end{itemize}

\column{0.45\textwidth}
\begin{block}{¿Cuánto creerle a una mejora?}
\small
\begin{itemize}
  \item $F_1$ \textbf{honesto}: optimismo $\leq$ 0,0005.
  \item \textbf{Otras particiones}: 0,7775 / 0,7784 / 0,7767.
  \item Kaggle público = 20\% de test, \alert{DE $\approx$ 0,006}: no
    distingue mejoras de 0,001. \textbf{Se decide por CV.}
\end{itemize}
\end{block}
\end{columns}
\end{frame}

\begin{frame}{Optimización: hiperparámetros y funciones de pérdida \hfill \pher}
\begin{columns}[T]
\column{0.4\textwidth}
\footnotesize
\textbf{Clasificador por hogar}\\[1mm]
\begin{tabular}{@{}ll@{}}
\toprule
\texttt{eta} (learning rate) & 0,02 \\
\texttt{max\_depth} & 8 \\
\texttt{min\_child\_weight} & 50 \\
\texttt{colsample\_bytree} & 0,5 \\
\texttt{subsample} & 0,8 \\
árboles (parada temprana) & 613--869 \\
\texttt{gamma} / \texttt{lambda} / \texttt{alpha} & 0 / 1 / 0 \\
\bottomrule
\end{tabular}

\vspace{1mm}
Superficie plana (0,772--0,773). \texttt{lambda} (L2) y \texttt{alpha} (L1)
\textbf{fijos} desde iteraciones previas, para aislar lo nuevo.

\column{0.57\textwidth}
\small
\textbf{1. Ingreso por persona con Tweedie} ($\rho = 1{,}2$; laboral y no
laboral)
\par\vspace{1mm}
{\centering $L(y,\mu) = -\dfrac{y\,\mu^{1-\rho}}{1-\rho} + \dfrac{\mu^{2-\rho}}{2-\rho}$\par}
\vspace{1mm}
\footnotesize Predice la \emph{media} en pesos (no la mediana) y admite
ceros: $R^2$ del hogar 0,63 $\to$ 0,66; $F_1$ 0,7733 $\to$ \textbf{0,7772}.

\vspace{1mm}
\small
\textbf{2. Pesos en la regresión}, $z=\log(\text{ingreso}/\text{línea})$:
\par\vspace{1mm}
{\centering $w_i = 1 + 3\,e^{-|z_i|/0{,}25}$\par}
\vspace{1mm}
\footnotesize Más peso cerca de la línea, de los dos lados. Final:
$p = 0{,}7\,p_{\text{clasif}} + 0{,}3\,p_{\text{regr}}$; pobre si $p>0{,}39$.
\end{columns}
\end{frame}

\section{Implicaciones de política}

\begin{frame}{Del $F_1$ a la focalización: leakage vs.\ undercoverage}
\begin{columns}[T]
\column{0.42\textwidth}
\small
Umbral 0,39, predicciones fuera de muestra:
\begin{itemize}
  \item \textbf{Cobertura} $\approx$81\% $\Rightarrow$
    \alert{undercoverage} $\approx$19\%.
  \item \textbf{Precisión} $\approx$75\% $\Rightarrow$ \alert{leakage}: 1 de
    cada 4 beneficiarios no es pobre.
  \item El $F_1$ pesa igual los dos errores; un programa no tiene por qué:
    el umbral es una \textbf{decisión de política}.
\end{itemize}
\footnotesize
Rural, informal y hogares grandes: más \emph{inclusión}. Pobres ``que no lo
parecen'' (jefe formal o educado, hogar pequeño): más \emph{exclusión}. Jefa
mujer: +3,8 pp de detección. Son tasas base distintas, no una regla
discriminatoria (Kleinberg et al., 2017).

\begin{frame}{Pobreza monetaria, focalización y costo de operación}
\begin{columns}[T]
\column{0.48\textwidth}
\small
\textbf{Qué mide el modelo}
\begin{itemize}
  \item Solo \textbf{pobreza monetaria}: educación, salud y subsidios casi
    no aportan una vez se conoce el trabajo.
  \item IPM y pobreza monetaria capturan privaciones distintas (Angulo et
    al., 2016): focalizar ingreso con indicadores de salud o educación
    genera errores.
  \item Proxies observables son manipulables (Camacho y Conover, 2011).
\end{itemize}

\column{0.5\textwidth}
\small
\textbf{¿Encuesta corta y barata?}
\begin{itemize}
  \item[\textcolor{verde}{+}] No pregunta \textbf{montos de ingreso} (lo más
    caro y subdeclarado): basta el módulo laboral, la vivienda y fuentes de
    ingreso sí/no.
  \item[\textcolor{rojo}{--}] Necesita el módulo laboral de \emph{cada
    miembro} ($\sim$60 preguntas por persona): costo similar a una visita
    SISBÉN.
  \item[\textcolor{rojo}{--}] Formalidad verificable con PILA; ingresos no
    laborales y búsqueda de trabajo, no.
  \item[\textcolor{gris}{$\bullet$}] Reentrenar si cambian la línea o el
    mercado laboral (datos de 2018).
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}{Cierre: qué nos hizo ganar}
\begin{columns}[T]
\column{0.32\textwidth}
\begin{block}{\pval}
\small CV anidada sin fuga, folds por hogar, $F_1$ honesto y ruido medido.
Decidimos con la CV, no con el leaderboard.
\end{block}
\column{0.32\textwidth}
\begin{block}{\pher}
\small Categóricas nativas, pérdida Tweedie, stacking con modelos por persona

\end{block}
\column{0.32\textwidth}

\small La pobreza es del hogar; el arriendo imputado reproduce la regla del

\end{block}
\end{columns}
