\section{Datos}
% =============================================================================

\begin{frame}{Datos: limpieza y nombres que se pueden leer}
\begin{columns}[T]
\column{0.4\textwidth}
\small
GEIH 2018: \textbf{543.103 personas, 164.958 hogares}; 20,0\% de hogares
pobres.
\begin{itemize}\footnotesize
  \item \texttt{pobre} = ingreso per cápita (con arriendo imputado) $<$
    línea: se cumple en el 100\% de los hogares.
  \item Los \textbf{montos de ingreso} no están en test: se eliminan de
    train.
  \item \texttt{NA} = ``la pregunta no aplica'' (saltos del cuestionario);
    no se reemplaza por 0 sin revisar.
  \item Sí/no a 1/0; categóricas con su código DANE \emph{y} una dummy
    por categoría.
  \item Se eliminan 2 hogares de train con cuartos imposibles; de test,
    nunca.
  \item Base por hogar: sumas, proporciones y máximos de los miembros.
\end{itemize}

\begin{columns}[T]

\begin{tabular}{@{}p{4.2cm}cc@{}}
\toprule

Se explica a un beneficiario & \textcolor{verde}{sí} & \textcolor{rojo}{no} \\
Auditable por un ente de control & \textcolor{verde}{sí} & \textcolor{rojo}{difícil} \\
Costo de mantener y reentrenar & \textcolor{verde}{bajo} & \textcolor{rojo}{alto} \\
Riesgo de manipulación & \textcolor{rojo}{mayor} & \textcolor{verde}{menor} \\
Captura no linealidades & \textcolor{rojo}{a mano} & \textcolor{verde}{sola} \\
\bottomrule
\end{tabular}

\vspace{2mm}
\footnotesize Un instrumento de focalización tiene que ser \textbf{creíble}:
los hogares y los entes de control deben entender por qué alguien entra o
sale (Camacho y Conover, 2011; Kleinberg et al., 2015).

\column{0.47\textwidth}
\begin{block}{Recomendación de política}
\small
\begin{enumerate}\footnotesize
  \item \textbf{Instrumento oficial: un logit parsimonioso} (15--40
    variables del módulo laboral y la vivienda), publicable y
    discutible.
  \item \textbf{XGBoost como auditor}: señala los hogares que el logit
    excluiría y que probablemente son pobres (los que ``no lo parecen''),
    para verificarlos en campo.
  \item Elegir el \textbf{umbral} según el costo social de excluir vs.\
    incluir, no por $F_1$.
\end{enumerate}
\end{block}
\vspace{1mm}
\footnotesize\textbf{Para discutir:} ¿cuántos puntos de $F_1$ vale la
transparencia? ¿quién responde por el error de una caja negra?
\end{columns}
\end{frame}
